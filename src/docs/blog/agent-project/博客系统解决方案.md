# 🎯 博客系统整体解决方案

基于 **Vue 3 + TypeScript + Vite** 技术栈，实现 Markdown 博客文章的**分类、标签、渲染**功能。

## 📋 方案架构

### 1. 技术选型

| 功能 | 技术方案 | 说明 |
|------|---------|------|
| MD 解析 | **markdown-it** | 功能强大、可扩展 |
| 代码高亮 | **highlight.js** | 支持多种语言 |
| 前置元数据 | **gray-matter** | 解析 YAML frontmatter |
| MD 文件导入 | **Vite import.meta.glob** | 静态分析、按需加载 |

### 2. Markdown 工具函数（src/utils/markdown.ts）

```typescript
import MarkdownIt from 'markdown-it'
import hljs from 'highlight.js'
import matter from 'gray-matter'

// 配置 markdown-it
const md = new MarkdownIt({
  html: true,
  linkify: true,
  typographer: true,
  highlight: (str, lang) => {
    if (lang && hljs.getLanguage(lang)) {
      try {
        return hljs.highlight(str, { language: lang }).value
      } catch (__) {}
    }
    return ''
  }
})

/**
 * 解析 Markdown 文件
 */
export function parseMarkdown(content: string) {
  const { data, content } = matter(content)
  const html = md.render(content)

  return {
    meta: data,
    content: html,
    raw: content
  }
}

/**
 * 从文件路径生成文章 ID
 */
export function generateIdFromPath(path: string): string {
  return path
    .replace(/^.*\/docs\//, '')
    .replace(/\.md$/, '')
    .replace(/\//g, '-')
}
```

### 3. 博客数据处理（src/composables/useBlog.ts）

```typescript
import { ref, computed } from 'vue'
import type { BlogPost, BlogMeta, Category, Tag } from '@/types/blog'

// 导入所有 Markdown 文件
const blogModules = import.meta.glob('@/docs/**/*.md', {
  query: '?raw',
  import: 'default'
})

export function useBlog() {
  const posts = ref<BlogPost[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  /**
   * 加载所有博客文章
   */
  async function loadAllPosts() {
    loading.value = true
    error.value = null

    try {
      const loadedPosts: BlogPost[] = []

      for (const path in blogModules) {
        const content = await blogModules[path]() as string
        const { parseMarkdown, generateIdFromPath } = await import('@/utils/markdown')

        const { meta, content: html } = parseMarkdown(content)
        const id = generateIdFromPath(path)

        loadedPosts.push({
          id,
          path,
          meta: meta as BlogMeta,
          content: html
        })
      }

      // 按日期排序（最新的在前）
      posts.value = loadedPosts.sort((a, b) =>
        new Date(b.meta.date).getTime() - new Date(a.meta.date).getTime()
      )
    } catch (e) {
      error.value = e instanceof Error ? e.message : '加载失败'
    } finally {
      loading.value = false
    }
  }

  /**
   * 获取所有分类
   */
  const categories = computed<Category[]>(() => {
    const categoryMap = new Map<string, number>()

    posts.value.forEach(post => {
      const cat = post.meta.category
      categoryMap.set(cat, (categoryMap.get(cat) || 0) + 1)
    })

    return Array.from(categoryMap.entries()).map(([name, count]) => ({
      name,
      count,
      slug: name.toLowerCase().replace(/\s+/g, '-')
    }))
  })

  /**
   * 获取所有标签
   */
  const tags = computed<Tag[]>(() => {
    const tagMap = new Map<string, number>()

    posts.value.forEach(post => {
      post.meta.tags.forEach(tag => {
        tagMap.set(tag, (tagMap.get(tag) || 0) + 1)
      })
    })

    return Array.from(tagMap.entries()).map(([name, count]) => ({
      name,
      count
    }))
  })

  /**
   * 根据分类筛选文章
   */
  function filterByCategory(category: string) {
    return posts.value.filter(post =>
      post.meta.category.toLowerCase() === category.toLowerCase()
    )
  }

  /**
   * 根据标签筛选文章
   */
  function filterByTag(tag: string) {
    return posts.value.filter(post =>
      post.meta.tags.some(t => t.toLowerCase() === tag.toLowerCase())
    )
  }

  /**
   * 根据 ID 获取文章
   */
  function getPostById(id: string) {
    return posts.value.find(post => post.id === id)
  }

  return {
    posts,
    loading,
    error,
    categories,
    tags,
    loadAllPosts,
    filterByCategory,
    filterByTag,
    getPostById
  }
}
```

### 4. BlogMain.vue 组件实现

```vue
<script setup lang="ts">
import { ref, onMounted, computed } from 'vue'
import { useBlog } from '@/composables/useBlog'
import BlogCard from './BlogCard.vue'
import type { BlogPost } from '@/types/blog'

const {
  posts,
  loading,
  error,
  categories,
  tags,
  loadAllPosts,
  filterByCategory,
  filterByTag
} = useBlog()

// 筛选状态
const selectedCategory = ref<string>('all')
const selectedTag = ref<string>('')

// 显示的文章列表
const displayedPosts = computed<BlogPost[]>(() => {
  let result = posts.value

  if (selectedCategory.value !== 'all') {
    result = filterByCategory(selectedCategory.value)
  }

  if (selectedTag.value) {
    result = result.filter(post =>
      post.meta.tags.some(t => t === selectedTag.value)
    )
  }

  return result
})

// 选择分类
function selectCategory(category: string) {
  selectedCategory.value = category
  selectedTag.value = '' // 重置标签
}

// 选择标签
function selectTag(tag: string) {
  selectedTag.value = tag
}

onMounted(() => {
  loadAllPosts()
})
</script>

<template>
  <div class="max-w-4xl mx-auto p-6">
    <!-- 筛选栏 -->
    <div class="mb-8 flex flex-wrap gap-4 items-center">
      <!-- 分类筛选 -->
      <div class="flex gap-2 items-center">
        <span class="text-gray-600 font-medium">分类:</span>
        <button
          @click="selectCategory('all')"
          :class="[
            'px-3 py-1 rounded-full text-sm transition',
            selectedCategory === 'all'
              ? 'bg-blue-500 text-white'
              : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
          ]"
        >
          全部 ({{ posts.length }})
        </button>
        <button
          v-for="cat in categories"
          :key="cat.slug"
          @click="selectCategory(cat.slug)"
          :class="[
            'px-3 py-1 rounded-full text-sm transition',
            selectedCategory === cat.slug
              ? 'bg-blue-500 text-white'
              : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
          ]"
        >
          {{ cat.name }} ({{ cat.count }})
        </button>
      </div>
    </div>

    <!-- 标签云 -->
    <div v-if="tags.length > 0" class="mb-8">
      <h3 class="text-gray-600 font-medium mb-3">标签:</h3>
      <div class="flex flex-wrap gap-2">
        <button
          @click="selectTag('')"
          :class="[
            'px-3 py-1 rounded-full text-sm transition',
            !selectedTag
              ? 'bg-purple-500 text-white'
              : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
          ]"
        >
          全部
        </button>
        <button
          v-for="tag in tags"
          :key="tag.name"
          @click="selectTag(tag.name)"
          :class="[
            'px-3 py-1 rounded-full text-sm transition',
            selectedTag === tag.name
              ? 'bg-purple-500 text-white'
              : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
          ]"
        >
          {{ tag.name }} ({{ tag.count }})
        </button>
      </div>
    </div>

    <!-- 加载状态 -->
    <div v-if="loading" class="text-center py-12 text-gray-500">
      加载中...
    </div>

    <!-- 错误提示 -->
    <div v-else-if="error" class="text-center py-12 text-red-500">
      {{ error }}
    </div>

    <!-- 文章列表 -->
    <div v-else-if="displayedPosts.length > 0">
      <BlogCard
        v-for="post in displayedPosts"
        :key="post.id"
        :post="post"
      />
    </div>

    <!-- 空状态 -->
    <div v-else class="text-center py-12 text-gray-500">
      没有找到相关文章
    </div>
  </div>
</template>
```

## 🎯 核心功能说明

### 1. **自动导入 MD 文件**
使用 Vite 的 `import.meta.glob` 功能，自动扫描 `src/docs` 目录下的所有 `.md` 文件。

### 2. **Frontmatter 解析**
通过 `gray-matter` 解析 MD 文件顶部的 YAML 元数据（标题、日期、分类、标签等）。

### 3. **分类筛选**
- 根据 `category` 字段自动统计文章数量
- 支持按分类筛选文章列表

### 4. **标签云**
- 根据 `tags` 数组自动聚合所有标签
- 支持按标签筛选文章
- 显示每个标签的文章数量

### 5. **Markdown 渲染**
- 使用 `markdown-it` 将 MD 转换为 HTML
- 集成 `highlight.js` 实现代码高亮

## 🚀 使用步骤

1. **安装依赖**
   ```bash
   npm install markdown-it highlight.js gray-matter
   ```

2. **创建文件**
   - 按照上述代码创建对应的类型、工具函数、组件文件

3. **编写 MD 文章**
   - 在 `src/docs` 目录下创建分类文件夹
   - 每个 MD 文件包含 frontmatter 元数据

4. **启动项目**
   ```bash
   npm run dev
   ```
