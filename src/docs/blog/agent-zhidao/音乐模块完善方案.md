## 二、模块定位与功能设计

### 2.1 音乐模块是什么？

**音乐模块** 是个人博客的音频内容展示和播放系统，具有以下特点：

| 特征 | 说明 |
|------|------|
| 🎵 **音乐展示** | 展示博主推荐的音乐、歌单 |
| 🎧 **在线播放** | 支持 MP3/FLAC 等格式在线播放 |
| 📝 **歌词同步** | 支持 LRC 歌词滚动显示 |
| 🎨 **可视化效果** | 音频波形、频谱可视化 |
| 📋 **歌单管理** | 创建和管理个人歌单 |
| 🔀 **播放控制** | 播放/暂停/上一曲/下一曲/循环 |

### 2.2 核心价值

- ✅ **展示品味** - 展示博主的音乐品味
- ✅ **增强体验** - 为访问者提供音乐陪伴
- ✅ **放松氛围** - 营造轻松的浏览环境
- ✅ **个性化** - 定制专属音乐空间

### 2.3 与其他模块的关系

| 模块 | 关系 | 说明 |
|------|------|------|
| **博客** | 独立 | 博客阅读时可播放背景音乐 |
| **说说** | 独立 | 可在说说中分享音乐 |
| **游戏** | 配合 | 游戏时可播放背景音乐 |

#### MusicPlayer.vue（播放器主组件）

**职责**：
- 管理音频播放
- 显示当前音乐信息
- 整合控制条、进度条、音量等
- 集成歌词面板

**核心逻辑**：
```typescript
import { ref, onMounted, onUnmounted } from 'vue'
import { useMusicStore } from '@/stores/music'

export function useMusicPlayer() {
  const audio = ref<HTMLAudioElement | null>(null)
  const store = useMusicStore()

  // 初始化音频对象
  onMounted(() => {
    audio.value = new Audio()
    setupAudioEvents()
  })

  // 设置音频事件监听
  function setupAudioEvents() {
    if (!audio.value) return

    audio.value.addEventListener('timeupdate', handleTimeUpdate)
    audio.value.addEventListener('ended', handleEnded)
    audio.value.addEventListener('loadedmetadata', handleLoadedMetadata)
    audio.value.addEventListener('error', handleError)
    audio.value.addEventListener('progress', handleProgress)
  }

  // 播放/暂停
  async function togglePlay() {
    if (!audio.value) return

    if (store.status === 'playing') {
      audio.value.pause()
      store.setStatus('paused')
    } else {
      await audio.value.play()
      store.setStatus('playing')
    }
  }

  // 加载音乐
  async function loadMusic(music: Music) {
    if (!audio.value) return

    store.setStatus('loading')
    audio.value.src = music.url
    audio.value.load()
  }

  // 跳转到指定时间
  function seek(time: number) {
    if (!audio.value) return
    audio.value.currentTime = time
  }

  // 设置音量
  function setVolume(volume: number) {
    if (!audio.value) return
    audio.value.volume = volume
  }

  return {
    audio,
    togglePlay,
    loadMusic,
    seek,
    setVolume
  }
}
```

```typescript
import { ref, watch, computed } from 'vue'
import type { LyricLine } from '@/types/music'

export function useLyric(lyrics: LyricLine[], currentTime: number) {
  const containerRef = ref<HTMLElement | null>(null)

  // 当前高亮的歌词行索引
  const currentIndex = computed(() => {
    return lyrics.findIndex((line, idx) => {
      const nextLine = lyrics[idx + 1]
      return (
        currentTime >= line.time &&
        (!nextLine || currentTime < nextLine.time)
      )
    })
  })

  // 自动滚动到当前行
  function scrollToCurrent() {
    if (!containerRef.value) return

    const currentLine = containerRef.value.querySelector('.lyric-line.active')
    if (currentLine) {
      currentLine.scrollIntoView({
        behavior: 'smooth',
        block: 'center'
      })
    }
  }

  return {
    containerRef,
    currentIndex,
    scrollToCurrent
  }
}
```

```typescript
import { ref, onMounted, onUnmounted } from 'vue'

export function useAudioVisualizer(audio: HTMLAudioElement) {
  const canvasRef = ref<HTMLCanvasElement | null>(null)
  let audioContext: AudioContext | null = null
  let analyser: AnalyserNode | null = null
  let animationId: number | null = null

  function initVisualizer() {
    audioContext = new (window.AudioContext || (window as any).webkitAudioContext)()
    analyser = audioContext.createAnalyser()

    const source = audioContext.createMediaElementSource(audio)
    source.connect(analyser)
    analyser.connect(audioContext.destination)

    analyser.fftSize = 256
    draw()
  }

  function draw() {
    if (!canvasRef.value || !analyser) return

    const canvas = canvasRef.value
    const ctx = canvas.getContext('2d')!
    const bufferLength = analyser.frequencyBinCount
    const dataArray = new Uint8Array(bufferLength)

    analyser.getByteFrequencyData(dataArray)

    // 绘制逻辑...
    animationId = requestAnimationFrame(draw)
  }

  return {
    canvasRef,
    initVisualizer
  }
}
```

### 5.1 状态管理（Pinia Store）

**文件**: `src/stores/music.ts`

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { Music, Playlist, PlayerState, PlayMode, PlayerStatus } from '@/types/music'
import { musicData, playlistData } from '@/config/music/music.data'

export const useMusicStore = defineStore('music', () => {
  // ========== 状态 ==========
  const playerState = ref<PlayerState>({
    currentMusic: null,
    currentPlaylist: null,
    status: 'idle',
    currentTime: 0,
    volume: 0.7,
    isMuted: false,
    playMode: 'sequence',
    playList: [],
    currentIndex: -1,
    loadProgress: 0
  })

  const playlists = ref<Playlist[]>(playlistData)
  const allMusic = ref<Music[]>(musicData)

  // ========== 计算属性 ==========

  /** 总音乐数 */
  const totalMusic = computed(() => allMusic.value.length)

  /** 总歌单数 */
  const totalPlaylists = computed(() => playlists.value.length)

  /** 当前播放列表的总时长 */
  const currentPlaylistDuration = computed(() => {
    return playerState.value.playList.reduce((sum, music) => sum + music.duration, 0)
  })

  /** 格式化当前时间 */
  const formattedCurrentTime = computed(() => {
    return formatTime(playerState.value.currentTime)
  })

  /** 格式化总时长 */
  const formattedDuration = computed(() => {
    const music = playerState.value.currentMusic
    return music ? formatTime(music.duration) : '0:00'
  })

  /** 进度百分比 */
  const progressPercent = computed(() => {
    const music = playerState.value.currentMusic
    if (!music) return 0
    return (playerState.value.currentTime / music.duration) * 100
  })

  // ========== 方法 ==========

  /** 加载歌单 */
  function loadPlaylist(playlist: Playlist) {
    playerState.value.currentPlaylist = playlist
    playerState.value.playList = allMusic.value.filter(m =>
      playlist.musicIds.includes(m.id)
    )
    playerState.value.currentIndex = 0

    if (playerState.value.playList.length > 0) {
      playMusic(playerState.value.playList[0])
    }
  }

  /** 播放音乐 */
  function playMusic(music: Music) {
    playerState.value.currentMusic = music
    playerState.value.status = 'loading'
    // 触发音频加载...
  }

  /** 播放/暂停 */
  function togglePlay() {
    const status = playerState.value.status
    if (status === 'playing') {
      playerState.value.status = 'paused'
    } else {
      playerState.value.status = 'playing'
    }
  }

  /** 下一曲 */
  function nextMusic() {
    const { currentIndex, playList, playMode } = playerState.value
    if (playList.length === 0) return

    let nextIndex = 0

    switch (playMode) {
      case 'sequence':
        nextIndex = (currentIndex + 1) % playList.length
        break
      case 'random':
        nextIndex = Math.floor(Math.random() * playList.length)
        break
      case 'loop-one':
        // 单曲循环，不切换
        return
      case 'loop':
        nextIndex = (currentIndex + 1) % playList.length
        break
    }

    playerState.value.currentIndex = nextIndex
    playMusic(playList[nextIndex])
  }

  /** 上一曲 */
  function prevMusic() {
    const { currentIndex, playList } = playerState.value
    if (playList.length === 0) return

    const prevIndex = currentIndex <= 0
      ? playList.length - 1
      : currentIndex - 1

    playerState.value.currentIndex = prevIndex
    playMusic(playList[prevIndex])
  }

  /** 切换播放模式 */
  function togglePlayMode() {
    const modes: PlayMode[] = ['sequence', 'loop', 'random', 'loop-one']
    const currentIndex = modes.indexOf(playerState.value.playMode)
    const nextIndex = (currentIndex + 1) % modes.length
    playerState.value.playMode = modes[nextIndex]
  }

  /** 更新当前时间 */
  function updateCurrentTime(time: number) {
    playerState.value.currentTime = time
  }

  /** 跳转进度 */
  function seekTo(percent: number) {
    const music = playerState.value.currentMusic
    if (!music) return

    playerState.value.currentTime = (percent / 100) * music.duration
    // 触发音频跳转...
  }

  /** 设置音量 */
  function setVolume(volume: number) {
    playerState.value.volume = Math.max(0, Math.min(1, volume))
    playerState.value.isMuted = volume === 0
  }

  /** 切换静音 */
  function toggleMute() {
    playerState.value.isMuted = !playerState.value.isMuted
    playerState.value.volume = playerState.value.isMuted ? 0 : 0.7
  }

  /** 更新加载进度 */
  function updateLoadProgress(progress: number) {
    playerState.value.loadProgress = progress
  }

  return {
    // 状态
    playerState,
    playlists,
    allMusic,

    // 计算属性
    totalMusic,
    totalPlaylists,
    currentPlaylistDuration,
    formattedCurrentTime,
    formattedDuration,
    progressPercent,

    // 方法
    loadPlaylist,
    playMusic,
    togglePlay,
    nextMusic,
    prevMusic,
    togglePlayMode,
    updateCurrentTime,
    seekTo,
    setVolume,
    toggleMute,
    updateLoadProgress
  }
}, {
  persist: {
    key: 'blog_music_store',
    storage: localStorage,
    pick: ['volume', 'playMode']
  }
})
```

### 5.2 歌词解析

```typescript
import type { LyricLine } from '@/types/music'

/**
 * 解析 LRC 歌词
 */
export function parseLRC(lrc: string): LyricLine[] {
  const lines: LyricLine[] = []
  const timeRegex = /^\[(\d{2}):(\d{2})\.(\d{2,3})\](.*)$/

  lrc.split('\n').forEach(line => {
    const match = line.match(timeRegex)
    if (match) {
      const minutes = parseInt(match[1])
      const seconds = parseInt(match[2])
      const milliseconds = parseInt(match[3].padEnd(3, '0'))
      const text = match[4].trim()

      lines.push({
        time: minutes * 60 + seconds + milliseconds / 1000,
        text
      })
    }
  })

  return lines
}

/**
 * 从 URL 加载歌词
 */
export async function loadLyric(url: string): Promise<LyricLine[]> {
  try {
    const response = await fetch(url)
    const lrc = await response.text()
    return parseLRC(lrc)
  } catch (error) {
    console.error('加载歌词失败:', error)
    return []
  }
}

/**
 * 搜索歌词
 */
export function searchLyric(lyrics: LyricLine[], keyword: string): LyricLine[] {
  if (!keyword.trim()) return lyrics

  const lowerKeyword = keyword.toLowerCase()
  return lyrics.filter(line =>
    line.text.toLowerCase().includes(lowerKeyword)
  )
}
```

```typescript
/**
 * 格式化时间（秒 -> mm:ss）
 */
export function formatTime(seconds: number): string {
  if (!isFinite(seconds) || seconds < 0) return '0:00'

  const mins = Math.floor(seconds / 60)
  const secs = Math.floor(seconds % 60)

  return `${mins}:${secs.toString().padStart(2, '0')}`
}

/**
 * 格式化详细时间（秒 -> HH:mm:ss）
 */
export function formatDetailedTime(seconds: number): string {
  if (!isFinite(seconds) || seconds < 0) return '0:00:00'

  const hours = Math.floor(seconds / 3600)
  const mins = Math.floor((seconds % 3600) / 60)
  const secs = Math.floor(seconds % 60)

  if (hours > 0) {
    return `${hours}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`
  }
  return `${mins}:${secs.toString().padStart(2, '0')}`
}

/**
 * 解析时间字符串（mm:ss -> 秒）
 */
export function parseTime(timeStr: string): number {
  const parts = timeStr.split(':').map(Number)

  if (parts.length === 2) {
    const [mins, secs] = parts
    return mins * 60 + secs
  } else if (parts.length === 3) {
    const [hours, mins, secs] = parts
    return hours * 3600 + mins * 60 + secs
  }

  return 0
}
```

```typescript
      case 'KeyL':
        if (e.ctrlKey) {
          toggleLyricPanel()
        }
        break
    }
  })
}
```

## 十四、常见问题解决

### 14.1 自动播放限制

**问题**：浏览器阻止自动播放音频

**原因**：浏览器策略，需要用户交互才能播放

**解决方案**：
```typescript
// 添加用户交互监听
function initAudio() {
  document.addEventListener('click', () => {
    if (audioContext && audioContext.state === 'suspended') {
      audioContext.resume()
    }
  }, { once: true })
}
```

### 14.2 CORS 跨域问题

**问题**：音频文件跨域无法播放

**解决方案**：
```typescript
// 方案一：配置服务器 CORS
// 服务器端添加响应头
// Access-Control-Allow-Origin: *

// 方案二：使用 crossOrigin
const audio = new Audio()
audio.crossOrigin = 'anonymous'
```

### 14.3 移动端播放中断

**问题**：切换应用或锁屏后播放中断

**解决方案**：
```typescript
// 使用 Page Visibility API
document.addEventListener('visibilitychange', () => {
  if (document.hidden) {
    // 页面隐藏，暂停播放
    pause()
  } else {
    // 页面显示，继续播放
    play()
  }
})
```
